cmake_minimum_required(VERSION 3.22)
project(shottime_voice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper-build)
add_library(shottime_whisper SHARED whisper-jni.cpp)
target_link_libraries(shottime_whisper PRIVATE whisper android log)
target_link_options(shottime_whisper PRIVATE "-Wl,-z,max-page-size=16384")

# The second build enables modern ARM kernels; Java checks CPU features before loading it.
if(GGML_CPU_ARM_ARCH STREQUAL "armv8.2-a+fp16+dotprod")
    set_target_properties(shottime_whisper PROPERTIES OUTPUT_NAME shottime_whisper_fast)
endif()
